% ------------------------------------------------------------
% Moldura temática das páginas de abertura dos capítulos
% ------------------------------------------------------------
%
% A moldura é aplicada somente à próxima página enviada ao PDF.
% A cor é expandida no momento da chamada do comando, evitando
% que um capítulo posterior altere a cor da moldura anterior.
%
\newcommand{\molduraTematica}{%
    \begingroup
        \edef\x{\endgroup
            \noexpand\AddToHookNext{shipout/background}{%
                \noexpand\begin{tikzpicture}[remember picture,overlay]
                    \noexpand\draw[
                        color=\corTema,
                        line width=0.45pt
                    ]
                    ([xshift=11mm,yshift=-11mm]current page.north west)
                    rectangle
                    ([xshift=-11mm,yshift=11mm]current page.south east);
                \noexpand\end{tikzpicture}%
            }%
        }%
    \x
}

% ------------------------------------------------------------
% Moldura neutra das páginas de abertura das grandes partes
% ------------------------------------------------------------
%
% As quatro partes representam grandes períodos da narrativa.
% Por isso utilizam uma moldura neutra, distinta das molduras
% temáticas dos capítulos.
%
\newcommand{\molduraParte}{%
    \begingroup
        \edef\x{\endgroup
            \noexpand\AddToHookNext{shipout/background}{%
                \noexpand\begin{tikzpicture}[remember picture,overlay]
                    \noexpand\draw[
                        color=CorCinza,
                        line width=0.35pt
                    ]
                    ([xshift=11mm,yshift=-11mm]current page.north west)
                    rectangle
                    ([xshift=-11mm,yshift=11mm]current page.south east);
                \noexpand\end{tikzpicture}%
            }%
        }%
    \x
}
