    font=small,

% Fotografia grande
\newcommand{\fotoGrande}[3]{%
    \begin{figure}[htbp]
        \centering
        \includegraphics[width=0.88\textwidth]{#1}
        \caption{#2}
        \label{#3}
    \end{figure}
}

% Fotografia média
\newcommand{\fotoMedia}[3]{%
    \begin{figure}[htbp]
        \centering
        \includegraphics[width=0.68\textwidth]{#1}
        \caption{#2}
        \label{#3}
    \end{figure}
}

% Fotografia pequena
\newcommand{\fotoPequena}[3]{%
